cmake_minimum_required(VERSION 3.24)
project(TaskbarLyricsHook LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "LyricsHook requires an x64 build")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
include(FetchContent)
# 固定 MinHook 1.3.4 的提交，避免构建时跟随远端分支变化。
FetchContent_Declare(minhook
  URL https://github.com/TsudaKageyu/minhook/archive/c3fcafdc10146beb5919319d0683e44e3c30d537.zip
  URL_HASH SHA256=cdcb160f734d81bd4d235dfea79e3f5a661c8ef0ab74fa814272aa5449069034)
FetchContent_MakeAvailable(minhook)
add_library(TaskbarLyricsHook SHARED Hook.cpp)
target_compile_features(TaskbarLyricsHook PRIVATE cxx_std_17)
target_link_libraries(TaskbarLyricsHook PRIVATE minhook gdiplus gdi32)
set_target_properties(TaskbarLyricsHook PROPERTIES OUTPUT_NAME TaskbarLyricsHook19)
add_executable(LyricsHookInjector Injector.cpp)
target_compile_features(LyricsHookInjector PRIVATE cxx_std_17)
add_executable(LyricsHookProbe Probe.cpp)
target_link_libraries(LyricsHookProbe PRIVATE gdiplus)
foreach(target TaskbarLyricsHook LyricsHookInjector LyricsHookProbe)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endforeach()
